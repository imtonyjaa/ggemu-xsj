//+++++++++++++++++++++++++++++++++
//-----隐藏人物 隐士鱼处理脚本-----
//+++++++++++++++++++++++++++++++++

//变量表：
//	Lany		-->	第几次和隐士鱼见面
//	TouchedWooy	-->	这次来看隐士鱼和他的第几次对话
//	RecentlyMeetWooy-->	最近见隐士鱼的次数，0 表示好久都没有见隐士鱼了
//	FollowWaitSide	-->	有谁在水流那侧等杨 0:没有同伴 1:纳兰真 2:月眉儿 3:紫轩 4:蔷薇 5:非女主角 >5 多于一个同伴
	
	Add($TouchedWooy, 1);
	If ($TouchedWooy >> 1) @TouchAgain;
	//又一次相会
	Add($Lany, 1);
	Add($RecentlyMeetWooy, 1);
	If ($Lany == 1)  @FirstMeet;
	If ($Lany == 2)  @SecondMeet;
	If ($Lany == 5)  @FifthMeet;
	If ($Lany == 7)  @SeventhMeet;
	If ($Lany == 9)  @NinethMeet;
	If ($Lany == 10) @TenthMeet;
	If ($Lany == 11) @EleventhMeet;
	Goto @OtherMeet;

@FirstMeet://鱼杨两人头次见面
	Say("杨影枫：这位仁兄好雅兴。", 2);
	Say("蓝衣小生：彼此彼此，兄台不也闲逛至此么。");
	Say("杨影枫：此处辟池引水，为鸟植林，为鱼凿潭。真是个好地方啊！", 2);
	Say("蓝衣小生：看这林中，鸟儿觅食相呼，观这潭里，水清出石鱼可数。确是一处怡人地方。小弟就居于附近，鄙姓<color=Blue>吴<color=Black>单名一个<color=Blue>越<color=Black>字，熟识的人称我隐士<color=Blue>鱼<color=Black>，常来此处逛逛，看公子面生，敢问姓氏？");
	Say("杨影枫：小弟姓<color=Blue>杨<color=Black>名<color=Blue>影枫<color=Black>，只是一匆匆过客，自幼习武，不想今日得识隐士，幸会幸会。", 2);
	Say("隐士鱼：哪里哪里。古人云<color=Red>小隐于野，中隐于市，大隐于朝。<color=Black>我小隐都算不上啊。");
	Say("杨影枫：吴兄此样逍遥自在，自不在乎大小。", 2);
	Say("隐士鱼：哈哈，杨兄真能言语。");
	Return;
@SecondMeet://鱼杨第二次见面
	Say("隐士鱼：东南方有一镇子，叫<color=Red>惠安镇<color=Black>，你可曾到过？");
	Say("杨影枫：到过，吴兄何事有此一问？", 2);
	Say("隐士鱼：以前我曾到过那里，见那镇子甚是繁华，现随便问问而已。");
	Say("杨影枫：是啊。小弟在那里还遇过些麻烦。", 2);
	Say("隐士鱼：哦。");
	Say("杨影枫：先是碰上了窃贼，被顺走了银钱。后又遇上歹人，害了一位新认识的兄台。", 2);
	Say("隐士鱼：杨兄可曾伤着？");
	Say("杨影枫：自己倒不曾伤着，只是被人误为那位兄台是我害的。幸得一友人相助，才了事。", 2);
	Say("隐士鱼：没有伤着就好，像我这般在郊野，就不会遇见这类事了。");
	If ($Event == 232) @Doubted;
	If ($Event >> 240) @Known;
	Say("杨影枫：城镇郊野各有其好处啊。", 2);
	Return;
	
	@Doubted://已经怀疑卓
		Say("杨影枫：只是……", 2);
		Say("隐士鱼：只是如何？");
		Say("杨影枫：听人言，那助我的友人却有些蹊跷。", 2);
		Say("隐士鱼：你可莫误会了好人。");
		If ($Event24Select == 0) @ChooseAgain;
		Return;
		@ChooseAgain://听了鱼的话后重新选择
			Choose("杨影枫想：","说话直言，还是得直问卓大哥。", "看来我还是暗访卓大哥吧。", $Event24Select);
			If ($Event24Select == 1) @Choose1;
			Return;
			@Choose1://改为暗访卓
				Add($EvilVal, -20);
				Return;
	@Known://已知真相
		Say("杨影枫：谁想最后方知那助我的友人却是幕后歹人。唉，世事为何如此？", 2);
		Say("隐士鱼：江湖纷争，大大小小，正正邪邪，何苦呢！");
		Say("杨影枫：江湖中也可实现些人生壮举，留名后世。", 2);
		Say("隐士鱼：你我想法相异，现在我想也劝你不动");
		Say("杨影枫：人各有志，我<color=Red>杨影枫<color=Black>还是乐于结交你这么个朋友。", 2);
		Return;
@FifthMeet://鱼杨第五次见面
	Say("杨影枫：我多次见你，总是独自一人，怎么也不见你有个佳人相伴？", 2);	
	Say("隐士鱼：佳人难觅啊！");
	Return;
@SeventhMeet://鱼杨第七次见面
	Say("杨影枫：吴兄近来可好？", 2);
	Say("隐士鱼：甚好。");
	Return;
@NinethMeet:
	Say("杨影枫：吴兄，别来无恙？", 2);
	Say("隐士鱼：托福托福。杨兄一向可好？");
	Say("杨影枫：你真是常来此处！遇见你好多次啦！", 2);
	Say("隐士鱼：是啊，此为我们第九次相遇了。");
	Say("杨影枫：九次！……吴兄真是好记性啊！", 2);
	Return;	
@TenthMeet://鱼杨第十次见面
	Say("杨影枫：我们又见面啦。", 2);
	Say("隐士鱼：我前几日得一<color=Red>紫霞玉佩<color=Black>，人言有调节经脉之效。你是习武之人，想来予你甚有好处，下次见面我带了送你。");
	GetGoodsNum("goods-n13-紫霞玉佩.ini");
	If ($GoodsNum >> 0) @AlreadyHave0;
	Return;
	@AlreadyHave0:
		Say("杨影枫：<color=Red>紫霞玉佩<color=Black>！我已有一块，恰好随身带着，你看看。", 2);
		Say("隐士鱼：和我的那块真是一摸一样！");
		Say("杨影枫：吴兄的好意心领了，看来你也不用专门带玉佩来给我。", 2);
		//跳过下次给玉佩的会面
		Add($Lany, 2);
		Return;
@EleventhMeet://鱼杨第十一次见面
	Say("隐士鱼：来来来，看我上次跟你提到的玉佩。");
	Say("杨影枫：此玉佩真是不错。", 2);
	GetGoodsNum("goods-n13-紫霞玉佩.ini");
	If ($GoodsNum >> 0) @AlreadyHave1;
		Say("隐士鱼：就这玉佩送给你了。");
		Say("杨影枫：这怎么好意思呢。", 2);
		Say("隐士鱼：你我也很熟了，就不用客气了。");
		Say("杨影枫：如此多谢吴兄。", 2);
		AddGoods("goods-n13-紫霞玉佩.ini");
		Return;	
	@AlreadyHave1:
		Say("杨影枫：我已有此样的玉佩了，你的好意我心领了。", 2);
		Say("隐士鱼：哦。");
		Say("杨影枫：你看我这块玉佩和你的可是一摸一样？", 2);
		Say("隐士鱼：确实一样，如此相象的两玉佩。");
		Add($Lany, 1);
		Return;
@OtherMeet://鱼杨的某次见面
	If ($Event == 170) @MeetAtEvent170;
	If ($Event == 185) @MeetAtEvent185;
	If ($RecentlyMeetWooy >> 1) @Continue2;
		Say("隐士鱼：杨兄，好久不见啦。");
		Say("杨影枫：是啊。", 2);
		Return;
	@Continue2:
		Say("隐士鱼：杨兄又来啦。");
		Say("杨影枫：来这可是总能遇见你啊。", 2);
		Return;
@MeetAtEvent170://带紫轩回寒波谷
	Say("隐士鱼：诶，杨兄，此次看你带个女伴来了，就在水流那侧。");
	Say("杨影枫：那是我新近结识的姑娘，她遇上了些麻烦，我带她到这来避避。", 2);
	Say("隐士鱼：原来如此。");
	Return;
@MeetAtEvent185://要去惠安镇查张仲天的事
	Say("杨影枫：吴兄，早啊。", 2);
	Say("隐士鱼：看你似乎要出门的样子。");
	Say("杨影枫：我正准备去<color=Red>慧安镇<color=Black>。", 2);
	Say("隐士鱼：看你象是有事要去那办。");
	Say("杨影枫：以前我和你提过，在慧安镇，我新认识的兄台遇上歹人被害之事。我想再去打听下消息，希望能查清此事情。", 2);
	Say("隐士鱼：那你小心了，希望你能早日查明。");
	Return;
@TouchAgain://此次相会的再次对话
	
	//过滤有特定对话的见面
	If ($Lany << 2 )   @ContinueTouch;
	If ($Lany == 7 )   @ContinueTouch;
	If ($Lany == 10)   @TenthMeetAgain;
	If ($Lany == 11)   @EleventhMeetAgain;
	
	If ($Event == 170) @TouchAtEvent170;
	If ($Event == 185) @TouchAtEvent185;
		Goto @ContinueTouch;
	@TouchAtEvent170:
		Say("隐士鱼：你快过去吧，别让那位姑娘久等了。");
		Return;
	@TouchAtEvent185:
		Say("隐士鱼：去<color=Red>惠安镇<color=Black>可有好一段路要走，你快动身吧。");
		Return;
	@ContinueTouch:
		If ($FollowWaitSide == 0) @Continue0;
		If ($TouchedWooy == 2) @SayBye;
			Goto @HaventGo;
		@Continue0:
			GetRandNum($Temp,0,1);
			If ($Temp == 0)	@TalkToWooy0;
				Say("隐士鱼：如此美景，你赋诗一首如何？");
				Say("杨影枫：我胸无点墨，就算了吧。", 2);
				Return;
			@TalkToWooy0:
				Say("隐士鱼：你闲时都作何消遣？");
				Say("杨影枫：我甚少消遣，时间多用来练习武功。", 2);
				Return;
	@TenthMeetAgain:
		Say("隐士鱼：下次我定记得带玉佩来给你。");
		Say("杨影枫：吴兄好意我已经很感激了。", 2);
		Return;
	@EleventhMeetAgain:
		Say("杨影枫：多谢吴兄！", 2);
		Say("隐士鱼：客气客气。");
		Return;
@SayBye:
	Say("杨影枫：还有朋友等我，下次有缘见面再叙，就此别过。", 2);
	Say("隐士鱼：杨兄走好。");
	Return;
@HaventGo://杨说要走，怎么还赖着不走
	If ($TouchedWooy == 2) @FirstBack;
	If ($TouchedWooy == 3) @ThirdBack;
	If ($TouchedWooy == 4) @FourthBack;
	If ($TouchedWooy << 7) @HaventYet;
		Say("隐士鱼：你无不无聊，还不走。");
		Return;
	@FirstBack://杨第一次回头
		Say("隐士鱼：杨兄何故不走了？");
		Say("杨影枫：得遇吴兄，真是想多聊一会儿啊。", 2);
		Say("隐士鱼：既有人等你，你还是快去吧。");
		Return;			
	@ThirdBack://杨第再次回头
		Say("隐士鱼：杨兄何故还未走？");
		Say("杨影枫：吴兄，我闲时再来寻你。", 2);
		Say("隐士鱼：好的，后会有期。");
		Return;
	@FourthBack://杨第四次回头
		Say("隐士鱼：杨兄该不会不知如何回去了吧？前方跃过水流，对面是一小径。");
		Say("杨影枫：哪里哪里，我正是循此路儿来。我这就走了，后会有期。", 2);
		Say("隐士鱼：后会有期。");
		Return;
	@HaventYet://快走快走，不走都要赶你走了。
		Say("影士鱼：杨兄如何还未走？");
		Say("杨影枫：这就走。", 2);
		Return;
