// 与与长安城卖菜人的对话

	CheckYear($Value);  
	If ($Value == 0) @NotYuanDan;	
	If ($Value == 1) @YuanDan;	
	
@NotYuanDan:

	If ($Talkmaicai == 0) @Talkmaicai0;
	If ($Talkmaicai == 1) @Talkmaicai1;
	If ($Talkmaicai == 2) @Talkmaicai2;
		
@Talkmaicai0:	
	Say("卖菜人：白菜青菜啊，青菜白菜啊，穷人菜当肉呀，富人肉当菜啦！", 0);
	Assign($Talkmaicai, 1);
	Return;

@Talkmaicai1:
	Say("独孤剑：老伯，我们在找一个叫李三的人，不知你见过没有？", 1);
	Say("卖菜人：你们找卖柴的李三？唉呀呀，不得了啊，那李三也真是条汉子，可惜就是不识时务，他居然敢管姚公子的事。那是粪坑里打灯笼——找死（屎）啊。", 0);
	Say("独孤剑：到底是怎么回事？", 8);
	Say("卖菜人：刚才姚公子在街上抢了个姑娘，被李三撞见。不想这李三上去打抱不平，那些打手就把他给抓到衙门大牢里去了，我看多半是有进无出啦！", 0);
	Say("张琳心：这姚公子是什么来头，怎么能随意抓人？", 27);
	Say("卖菜人：一听这话就知道你们是外地人。这长安城的衙门就是姚家的，你看这长安城也快成姚家的了。", 0);
	Say("张琳心：这话怎么讲？", 27);
	Say("卖菜人：这姚公子的老爹姚知节乃是长安府的父母官。他本是一个小吏，金人打来后，原来大宋的官们跑的跑，死的死，剩下一座空城。这姚知节，名知节实不知节，作了金人的走狗，当了长安城的知府。这姚公子又和金将花里花西拜了把子，父子俩就成了这长安城里的土皇帝了。这姚公子吃喝嫖赌无恶不作，看见漂亮的姑娘家就强行抢去糟蹋了……", 0);
	Say("独孤剑：这混账东西！他住在哪里？", 9);
	Say("张琳心：独孤哥哥，我们杀进衙门去救李三！", 21);
	Say("卖菜人：哎呀，使不得！使不得！二位千万不要去惹他，这姚公子权势薰天，又有一身武功，已经有好多人栽到他手里了。如果二位还怜惜我们这些老百姓的性命，更加万万使不得。", 0);
	Say("独孤剑：此话怎讲？", 8);
	Say("卖菜人：要是你们杀进衙门，金人拿不着你们，又要拿这里的百姓开刀。我们又是大难临头了。", 0);
	Say("张琳心：那你们就这样甘心忍受金人的凌辱？", 27);
	Say("卖菜人：姑娘，话虽如此。可在那个皇帝脚下，遭罪的一样是我们这些穷苦人。咱只求有个太平日子就行了。", 0);
	Say("张琳心：……", 28);
	Say("卖菜人：少管闲事吧，这样可以多活几年！", 0);
	Memo("独孤剑和张琳心来到长安。从路人处得知李三因路见不平，已经被恶少姚少爷抓走了。");
	Assign($Talkmaicai, 2);
	Assign($Event, 301);
	Return;
