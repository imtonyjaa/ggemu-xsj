//和药店掌柜对话

	CheckYear($Value);  
	If ($Value == 0) @NotYuanDan;	
	If ($Value == 1) @YuanDan;	
	
@NotYuanDan:

	Say("药店掌柜：本店金字招牌，诸药齐备，两位尽管挑选。",90);
	GetPlayerLevel($PlayerLevel);
	If ($PlayerLevel << 16) @PlayerLevelL16;
	If ($PlayerLevel << 31) @PlayerLevelL31;
	Goto @PlayerLevelG30;
	Return;

@PlayerLevelL16:
	SellGoods("低级药品.ini");
	Return;
@PlayerLevelL31:
	SellGoods("中级药品.ini");
	Return;
@PlayerLevelG30:
	SellGoods("高级药品.ini");
	Return;

	
@YuanDan:	

	Say("药店掌柜：新年好！祝公子新的一年财源广进，一本万利！本店金字招牌，诸药齐备，两位尽管挑选。",90);
	GetPlayerLevel($PlayerLevel);
	If ($PlayerLevel << 16) @PlayerLevelL16YD;
	If ($PlayerLevel << 31) @PlayerLevelL31YD;
	Goto @PlayerLevelG30YD;
	Return;

@PlayerLevelL16YD:
	SellGoods("低级药品.ini", 1);
	Return;
@PlayerLevelL31YD:
	SellGoods("中级药品.ini", 1);
	Return;
@PlayerLevelG30YD:
	SellGoods("高级药品.ini", 1);
	Return;
