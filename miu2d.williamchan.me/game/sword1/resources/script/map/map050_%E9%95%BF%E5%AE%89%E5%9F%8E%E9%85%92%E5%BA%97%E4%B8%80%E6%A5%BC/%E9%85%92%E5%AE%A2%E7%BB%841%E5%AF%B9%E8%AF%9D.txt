	If($Map050jiuke1 == 0) @Talk0;
	If($Map050jiuke1 == 1) @Talk1;
	If($Map050jiuke1 == 2) @Talk2;
	If($Map050jiuke1 == 3) @Talk3;
	If($Map050jiuke1 == 4) @Talk4;
	
@Talk0:
	Say("胡子：老弟，你可听说过《壶中谱》？");
	Say("胖子：可是五十年前洞庭湖上“湖岛醉仙”钟午子的不传医诀？");
	Say("胡子：正是，如果能得到这份医诀，李将军的绝症或许有救。");
	Assign($Map050jiuke1,1);
	Return;
	
@Talk1:
	Say("胡子：传说钟午子平生有两大绝学——“倒虚天”神功和《壶中谱》……");
	Say("胖子：可惜学会这绝世医学的就只有他的三弟子道一。");
	Say("胡子：这些徒弟只知杀人之法，却不学救人之术……焉不知杀千人易，救一人难啊！");
	Assign($Map050jiuke1,2);
	Return;
	
@Talk2:
	Say("胖子：这道一一死，世上恐怕再无人能将这神奇医术传扬下去……");
	Say("胡子：他的两个弟子，“药仙”古柏和“丹王”妙松不都是当世神医吗？");
	Say("胖子：只可惜，他俩各偏一道，未能融而通贯……");
	Assign($Map050jiuke1,3);
	Return;
	
@Talk3:
	Say("胡子：李将军这次怕是大限已到，我看撑不过今年了。");
	Say("胖子：诸王子各藏私心，西夏国恐怕又有一场大变……");
	Assign($Map050jiuke1,4);
	Return;
	
@Talk4:
	Say("胖子：客官欲知后事，且听“剑三”分解。");
	Return;
