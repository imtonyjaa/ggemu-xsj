//和菱珠对话

	If($Clue << 31) @Clue30;
	If($Clue >= 31) @Clue35;
	
@Clue30:
	If ($Talkdoujunnan == 1) @Talk1;
	Watch("窦俊男", "菱珠", 0);
	Say("窦俊男：阿珠，当年我和那黄脸婆是指腹为婚，岂料到今日我一朵鲜花插在牛粪中。");
	Say("窦俊男：我和祠堂的长老们都说好了，明天我就休了她……只要到明天，我俩就不必再这般偷偷摸摸了。");
	Say("菱珠：倘若她不愿意呢？");
	Say("窦俊男：她敢……");
	Say("窦俊男：那，那就让她休了我！");
	Say("菱珠：说来说去，你就是个没骨气的男人。");
	Assign($Talkdoujunnan, 1);
	Return;
@Talk1:
	Watch("窦俊男", "菱珠", 0);
	Say("菱珠：昊哥，不如我们现在就一起离开这里……");
	Say("窦俊男：好，不过……");
	Return;
	
@Clue35:
	If($Talkdoujunnan == 2) @Talk2;
	Watch("窦俊男", "菱珠", 0);
	Say("菱珠：昊哥，你怎么浑身……");
	Say("窦俊男：阿珠，我已经下定决心了……");
	Say("菱珠：真的！");
	Say("窦俊男：我们分手吧！");
	Say("菱珠：你——");
	Assign($Talkdoujunnan, 2);
	Return;
	
@Talk2:
	Watch("窦俊男", "菱珠", 0);
	Say("菱珠：想不到你……对你，我已经死心了。");
	Say("窦俊男：我，我——");
	Return;
