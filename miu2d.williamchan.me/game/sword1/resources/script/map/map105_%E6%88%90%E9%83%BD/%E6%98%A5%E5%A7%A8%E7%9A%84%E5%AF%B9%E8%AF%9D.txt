//和春姨对话

	If ($Clue << 31) @Clue30;
	If ($Clue >= 31) @Clue35;
	
@Clue30:
	If ($Talkchunyi == 1) @Talk1;
	
	Say("独孤剑：这位姑娘，请问……", 1);
	Say("春姨：公子——，找小女子有什么事啊？",0);
	Say("独孤剑：没，没有……", 6);
	Assign($Talkchunyi, 1);
	PlayerGotoDir(6,2);
	SetNpcDir("春姨",7);
	Return;
	
@Talk1:
	Say("春姨：看来眼角的粉还涂的不够……", 0);
	Say("春姨：这样是不是好多了！", 0);
	Return;
	
@Clue35:
	If ($Talksuyu == 1) @Talk2;
	Watch("春姨", "苏宇", 0);
	Say("春姨：这位公子，你为什么老盯着人家看？", 0);
	Say("苏宇：冤枉啊，在下哪里敢？");
	Say("春姨：你明明盯着小女子的背影看，还要抵赖。", 0);
	Say("苏宇：既然是看你的背影，你又如何得知？");
	Say("春姨：人家背上火辣辣的……", 0);
	Say("春姨：今天你不解释清楚，就别想走！", 0);
	Assign($Talksuyu, 1);
	Return;
	
@Talk2:
	Watch("春姨", "苏宇", 0);
	Say("春姨：说，你为什么盯着小女子看？", 0);
	Say("苏宇：救命啊——");
	Return;
