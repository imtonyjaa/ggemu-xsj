// 武夷派大弟子张林对话

	If ($Event == 30) @Event30; 
	If ($Event == 40) @Event40; 
	If ($Event == 220) @Event220;

@Event30:
	NpcSpecialAction("独孤剑", "mpc043.asf");
	Sleep(1500);
	Say("独孤剑：在下衡山剑派独孤剑，请问“沧浪剑客”柳老前辈是否在家？", 1);
	Say("张林：原来是衡山派的独孤师兄，我是张林，你找我师傅有什么事？", 0);
	Say("独孤剑：在下有要事求见柳老前辈。", 1);
	Say("张林：不巧得很，师傅前几天去了临安，已经不在山上了。", 0);
	Say("独孤剑：令师去了临安？", 1);
	Say("张林：是的，听说临安有不少奇怪的事情接连发生，师傅去看看，顺道拜访一位朋友。", 0);
	Say("独孤剑：在下有一封重要的信要亲手交给柳前辈。",1);
	Say("张林：如此的话，独孤兄得赶去临安一趟……", 0);
	SetNpcDir("张林", 0);
	Memo("独孤剑穿过九猴洞，来到武夷山顶，但是柳中原不在，从武夷派大弟子张林处得知柳中原去了临安，于是独孤剑下山前往临安。");
	StopMusic();
	PlayMusic("长路.mp3");
	Assign($Event, 40);
	Assign($Result, 0);
	Return;
	
@Event40:
	Say("张林：有劳独孤兄了。", 0);

@Event220:

	SetNpcActionFile("独孤剑", 0, "mpc007_2.asf");
	SetNpcActionFile("独孤剑", 1, "mpc007_2.asf");
	NpcSpecialAction("独孤剑", "mpc007_1.asf");    // 独孤剑蹲下
	Sleep(1500);
	Say("独孤剑：张兄！", 6);
	Say("张林：师傅退……退到北山的山洞中去了。那……那是本派……本派的禁地。", 0);
	Say("独孤剑：张兄你……", 6);
	Say("张林：快……快去救……救我师傅，他们山上有人埋伏，我……我……", 0);
	
	SetNpcActionFile("张林", 0, "mpc006_3.asf");
	SetNpcActionFile("张林", 1, "mpc006_3.asf");

	Say("独孤剑：张兄！张兄！", 6);
	Say("张琳心：他已经死了。", 24);
	
	SetNpcActionFile("独孤剑", 0, "npc001_st2.asf");
	SetNpcActionFile("独孤剑", 1, "npc001_sst2.asf");
	NpcSpecialAction("独孤剑", "mpc007_3.asf");    // 独孤剑站起	
	Sleep(1500);
	
	Say("独孤剑：看来敌人的计划很周密，我们要格外留心。", 9);
	Say("张琳心：（爹……）", 28);
	Say("张琳心：我们现在赶紧到禁地去看看！", 28);

	SetNpcScript("张琳心", "张琳心对话.txt");

	Memo("独孤剑和张琳心来到武夷山，遇到大量的五色教杀手。在大厅里找到尚未断气的武夷山大弟子张林，得知柳中原已经撤到禁地密洞，两人决定前往禁地。");
	Return;
