// 与住店客人郜广的对话

	If($Talkgaoguang == 0) @Talk0;
	If($Talkgaoguang == 1) @Talk1;
	If($Talkgaoguang == 2) @Talk2;
	
@Talk0:
	Say("郜广：看公子面相俊朗，英气迫人，将来必有造就。不知公子出自哪个门下？", 0);
	Say("独孤剑：晚辈衡山独孤剑，还请前辈不吝指教。", 1);
	Say("郜广：指教二字老夫岂敢受用？老夫在江湖中滚打四十年，却依然是个三四流角色，许多象你一般的后生小辈，武功已在老夫之上了，实在叫老夫汗颜。", 0);
	Say("郜广：老夫对江湖中事已深感厌倦，这次来临安消遣几日后便要金盆洗手退隐江湖了。", 0);
	Assign($Talkgaoguang, 1);
	Return;
	
@Talk1:	
	Say("郜广：老夫虽然武功平常，但对江湖中的事情却知之甚多，不知公子是否愿听？", 0);
	Say("独孤剑：能得到前辈的指点，乃是晚辈的造化，请前辈明示！", 1);
	Say("郜广：当今武林高手中堪称一流的，不过少林无虚大师，贵派的刘老爷子，武夷山“沧浪剑客”柳中原，华山大雁客顾枫，和“天心飞仙”四剑客中剩下的三位：天剑南宫灭，心剑方勉，飞剑张风。", 0);
	Say("独孤剑（暗忖）：师傅得到的名单上有其中的三个人。", 1);
	Say("郜广：无虚大师虽身在佛门，却时刻关注武林中事，为抗金大业出过不少力，不失大师风范。令师这些年倒象是很少在江湖中走动了，但是调教出了你这样的弟子，其他人在这一点就不如令师了。", 0); 
	Say("郜广：沧浪剑客柳中原和大雁客顾枫在江湖中走动频繁，但顾枫和其师弟林海多年来为争夺华山掌门而争斗不休，兄弟匿墙，可悲可叹。", 0);
	Say("郜广：从顾枫的为人看，他不会将掌门的位子看得很重，但始终不愿将这个位子让给林海，似有什么隐衷。小雁客林海心胸狭窄，争强好胜，顾枫恐怕是怕华山派的英名毁在了林海手里。", 0);
	Say("郜广：天剑南宫灭投效金国，残杀我大宋武林豪杰无数，江湖中人恨不生啖其肉，无奈其武功高超，奈他不何。前些年心剑方勉在泰山玉皇顶与其一战，将其打败，这些年他才有所收敛。", 0);
	Say("郜广：方勉因与南宫灭一战而名声大噪，江湖中人誉他为天下第一，他也坦然受之。只是自此之后再未见他出过手，平常也很难见到他，神龙见首不见尾。", 0);
	Say("郜广：至于飞剑张风，现官拜殿前都指挥使并掌管马步两路禁军，已是官场中走的多，江湖中走的少了。", 0);
	Say("独孤剑：请问郜老前辈，是否有五色教的消息？", 1);
	Say("郜广：五色教在三十多年前是武林中最大的邪教组织，专干杀人越货的勾当，人多势众，高手如云。武林中正派人士数次剿杀，均未成功，反而折损许多人手。", 0);
	Say("郜广：后来武夷山掌门柳四海组织各大门派组成血旗门，三十年前与五色教主海山阳相约，双方在武夷山上决一死战。双方大战数天，不分胜负，互有损伤。", 0);
	Say("郜广：后来柳四海将五色教人众诱入武夷山后的一个山洞里，结果进洞的人没有一个能出来，想来是双方在里面同归于尽了。只是从那以后武夷派就将此洞封为禁地，再也没人进去察看到底如何了。", 0);
	Say("郜广：从此以后，五色教就销声匿迹了。不过最近江湖中传说又有五色教的人在活动，说不定五色教经过三十年的休整，羽毛又渐渐丰满起来了。如果真是这样，武林又要面临一场浩劫了。", 0);
	Say("郜广：而且，听说南宫灭也在暗中网罗江湖高手，欲对我朝不利，唉，国运衰落，江湖恐慌，老夫还是归隐躬耕去罢。", 0);
	Assign($Talkgaoguang, 2);
	Return;
	
@Talk2:
	Say("郜广：唉，国运衰落，江湖恐慌，老夫还是归隐躬耕去罢。", 0);
	Return;
