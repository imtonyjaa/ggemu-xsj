// 和无智对话

	If ($Event >= 410) @Event410;

	If ($Talkwuzhi == 0) @Talkwuzhi0;
	If ($Talkwuzhi == 1) @Talkwuzhi1;
	
@Talkwuzhi0:
	Say("无智：阿弥陀佛！这位女施主，少林向来谢绝女客入访，你如何进来？", 0);
	Say("张琳心：又来了。独孤哥哥，你看我的。", 21);
	Say("张琳心：大师此言差矣！我且问大师：心生念，念生佛，无心无念，可曾有佛？", 21);
	Say("无智：阿弥陀佛！施主错了。佛自在自有，自枯自荣，岂是我等凡心所生！", 0);
	Say("张琳心：这个……那我再问大师：心可有男女之别，佛岂对众生不等。", 23);
	Say("无智：问得好。若在佛看来，自当众生无差。但我等皆为凡胎肉体，自然做不到心中无垢。少林所以门规森严，便是勉力修行之人拂去心中的尘垢，方能体悟我佛包容万物的境界。", 0);
	Say("张琳心：这个……", 23);
	Say("独孤剑：多谢大师指点。", 1);
	Say("无智：施主明白了吗？老衲还请施主速速离去。", 0);
	Say("张琳心：什么人……", 25);
	Sleep(500);
	SetNpcDir("无智", 4);
	Say("无智：咦？", 0);
	Sleep(500);
	Say("张琳心：独孤哥哥，快走！", 21);
	Say("独孤剑：啊？", 8);
	Assign($Talkwuzhi, 1);
	Return;
	
@Talkwuzhi1:	
	Say("无智：阿弥陀佛！施主原来在骗老衲。", 0);
	SetNpcDir("无智", 0);
	Return;
	
@Event410:
	Say("无智：怀慈悲心，达光明境界；怀平常心，致不二法门。", 0);
	Say("无智：南无阿弥陀佛！", 0);
	Return;
