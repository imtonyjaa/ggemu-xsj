	DisableNpcAI();
	SaveNpc();
	Say("独孤剑：这里好像是一个洞口，不过被巨石挡住了。", 1);
	Say("张琳心：这洞里说不定有什么秘密，我们何不一起用内力将它推开？", 21);

	If($TalkJiguan == 0) @Choose0;
	If($TalkJiguan == 1) @Choose1;

@Choose0:

	Say("独孤剑：好的，不过要小心，不知有什么机关。", 1);
	SetNpcActionFile("独孤剑", 0, "");
	SetNpcActionFile("独孤剑", 1, "");

	SaveObj();
	
	FadeOut();
	StopMusic();
	PlayMusic("黑影.mp3");
	Sleep(500);
	SaveNpc();
	ClearBody();
	SaveObj();
	LoadMap("map009_武夷山禁地内洞2.map");
	LoadNpc("map009.npc");
	LoadObj("map009.obj");
	SetNpcActionFile("独孤剑", 0, "npc001_st2.asf");
	SetNpcActionFile("独孤剑", 1, "npc001_sst2.asf");
