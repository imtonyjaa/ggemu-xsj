// 与酒楼老板的对话

	CheckYear($Value);  
	If ($Value == 0) @NotYuanDan;	
	If ($Value == 1) @YuanDan;	
	
@NotYuanDan:

	If ($Event == 40) @Event40;
	If ($Event == 50) @Event50; 
	If ($Event == 180) @Event180;

	Say("酒楼老板：客官要些什么，楼上有雅座。", 85);
	Return;
	
@Event40:

@Talk0:

	Say("酒楼老板：客官来点什么？", 85);
	Say("独孤剑：在下想打听一个人。", 1);
	Say("酒楼老板：公子请说，我这酒馆可是全京城的信息交换中心。", 85);
	Say("独孤剑：听说武夷山柳中原柳老前辈到了京城，不知在哪儿可以找到他老人家？", 1);
	Say("酒楼老板：你问我算是找到人了，柳老爷子前几天还和张大人来我这喝酒。这两天没来了，只怕是回去了。", 85);
	Say("独孤剑：多谢掌柜。", 2);
	Say("酒楼老板：要谢的话就多来喝几杯酒，照顾一下本店的生意，好不好？", 85);
	Memo("到酒楼里向老板打听柳中原的下落，得知他可能已经回武夷山。");
	Assign($JiuLvLaoBan, 1);
	Return;
